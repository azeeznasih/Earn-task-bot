// ============================================================ // 💰 MIN/MAX WITHDRAW SETTINGS — Logic // ============================================================ // Load all settings when page loads async function loadAllLimits() { try { const res = await apiCall('/miniapp/api/admin/settings'); if (res.success && res.settings) { const s = res.settings; // Withdraw document.getElementById('set_min_wd').value = s.min_withdraw || 10; document.getElementById('curr_min_wd').textContent = s.min_withdraw || 10; document.getElementById('set_max_wd').value = s.max_withdraw || 10000; document.getElementById('curr_max_wd').textContent = s.max_withdraw || 10000; document.getElementById('set_tax').value = s.tax_percent || 0; document.getElementById('curr_tax').textContent = s.tax_percent || 0; // Add Fund document.getElementById('set_min_af').value = s.auto_upi_min || 5; document.getElementById('curr_min_af').textContent = s.auto_upi_min || 5; document.getElementById('set_max_af').value = s.auto_upi_max || 200; document.getElementById('curr_max_af').textContent = s.auto_upi_max || 200; console.log('✅ Limits loaded:', s); } } catch (e) { console.error('Load limits error:', e); } } // Save single setting async function saveSetting(key, inputId) { const input = document.getElementById(inputId); const value = input.value.trim(); if (!value || isNaN(value)) { showToast('❌ Enter valid number'); haptic('error'); return; } const numValue = parseFloat(value); // Validation if (numValue < 0) { showToast('❌ Value must be positive'); haptic('error'); return; } // Special validation if (key === 'min_withdraw' && numValue < 1) { showToast('❌ Min withdraw must be at least ₹1'); haptic('error'); return; } if (key === 'max_withdraw') { const minW = parseFloat(document.getElementById('set_min_wd').value) || 10; if (numValue <= minW) { showToast('❌ Max must be greater than Min'); haptic('error'); return; } } if (key === 'tax_percent' && (numValue < 0 || numValue > 50)) { showToast('❌ Tax must be 0-50%'); haptic('error'); return; } try { const data = await apiCall('/miniapp/api/admin/settings/update', { method: 'POST', body: JSON.stringify({ key, value: numValue }) }); if (data.success) { // Update current display const currMap = { 'min_withdraw': 'curr_min_wd', 'max_withdraw': 'curr_max_wd', 'tax_percent': 'curr_tax', 'auto_upi_min': 'curr_min_af', 'auto_upi_max': 'curr_max_af' }; const currEl = document.getElementById(currMap[key]); if (currEl) currEl.textContent = numValue; // Visual feedback const btn = event?.target?.closest('.settings-btn.save'); if (btn) { btn.classList.add('saved'); btn.textContent = '✅ Saved'; setTimeout(() => { btn.classList.remove('saved'); btn.innerHTML = '💾 Save'; }, 1500); } showToast(`✅ ${key} = ${numValue}`); haptic('success'); // Reload all to ensure sync await loadAllLimits(); } else { showToast('❌ ' + (data.error || 'Failed')); haptic('error'); } } catch (e) { console.error('Save setting error:', e); showToast('❌ Network error'); haptic('error'); } } // Reset setting to default async function resetSetting(key, inputId, defaultValue) { if (!confirm(`Reset ${key} to ${defaultValue}?`)) return; const input = document.getElementById(inputId); input.value = defaultValue; try { const data = await apiCall('/miniapp/api/admin/settings/update', { method: 'POST', body: JSON.stringify({ key, value: defaultValue }) }); if (data.success) { // Update current display const currMap = { 'min_withdraw': 'curr_min_wd', 'max_withdraw': 'curr_max_wd', 'tax_percent': 'curr_tax', 'auto_upi_min': 'curr_min_af', 'auto_upi_max': 'curr_max_af' }; const currEl = document.getElementById(currMap[key]); if (currEl) currEl.textContent = defaultValue; showToast(`🔄 Reset to ${defaultValue}`); haptic('success'); await loadAllLimits(); } else { showToast('❌ ' + (data.error || 'Failed')); } } catch (e) { showToast('❌ Network error'); } } // Update switchTab to load limits when switched const originalSwitchTab = window.switchTab; window.switchTab = function(tab) { originalSwitchTab(tab); if (tab === 'limits') { loadAllLimits(); } }; // Load limits on page init document.addEventListener('DOMContentLoaded', () => { setTimeout(() => { loadAllLimits(); }, 1000); }); // Expose globals window.loadAllLimits = loadAllLimits; window.saveSetting = saveSetting; window.resetSetting = resetSetting; // ============================================================ // 🔄 ALSO: Update switchTab in original to load limits // ============================================================ // ഈ line കൂടി switchTab-ന്റെ അവസാനത്തിൽ add ചെയ്യൂ: // if (tab === 'limits') loadAllLimits();
